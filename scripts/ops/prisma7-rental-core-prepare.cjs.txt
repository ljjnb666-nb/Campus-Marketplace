const fs=require("node:fs"),ts=require("typescript"),path=require("node:path");
const [sourcePath,output,root]=process.argv.slice(2);
if(!sourcePath||!output||!root)throw Error("R3D02C01_INVALID_ARGUMENTS");
const source=fs.readFileSync(sourcePath,"utf8");
const ast=ts.createSourceFile(sourcePath,source,ts.ScriptTarget.Latest,true,ts.ScriptKind.TS);
if(ast.parseDiagnostics.length)throw Error("R3D02C01_SOURCE_PARSE");
const fns=[
 "writeStatusLog","isRentalOrderRoleParticipant","canCancelRentalOrder",
 "counterpartyId","approveRentalOrderTx","rejectRentalOrderTx",
 "confirmPickupTx","cancelRentalOrderTx"
];
const bodies=fns.map(name=>{
 const matches=ast.statements.filter(n=>ts.isFunctionDeclaration(n)&&n.name?.text===name);
 if(matches.length!==1||!matches[0].body)throw Error("R3D02C01_ENTRY_DRIFT_"+name);
 return source.slice(matches[0].getStart(ast),matches[0].end);
});
const imports=[
 'import type { Prisma, RentalOrderStatus, RentalCancellationReason } from "../generated/prisma/client.ts";',
 'import { prepareActiveAccountMutation } from "./lib/governance/active-account-mutation.ts";',
 'import { emitNotificationsTx } from "./lib/notifications/notification-service.ts";',
 'import { RENTAL_ORDER_APPROVED_KIND, RENTAL_ORDER_REJECTED_KIND, RENTAL_ORDER_CANCELLED_KIND, RENTAL_PICKUP_CONFIRMED_KIND } from "./lib/notifications/notification-registry.ts";'
];
fs.writeFileSync(output,imports.join("\n")+"\n"+bodies.join("\n\n")+"\n");
console.log("[worker-cycle] PASS R3D02C01 exact original rental approval/pickup/cancel AST declarations");
