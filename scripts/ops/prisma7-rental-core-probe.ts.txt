async function snapshot(id:string){
 const [order,logs,notes,handover,domain,outbox]=await Promise.all([
  root.rentalOrder.findUniqueOrThrow({where:{id}}),
  root.rentalOrderStatusLog.findMany({where:{orderId:id}}),
  root.notification.findMany({where:{kind:{in:kinds},dedupeKey:{contains:id}}}),
  root.rentalHandoverRecord.findUnique({where:{orderId:id}}),
  root.domainEvent.count({where:{aggregateId:id}}),
  root.outboxEvent.count({where:{aggregateId:id}})
 ]);
 return {order,logs,notes,handover,domain,outbox};
}
function singleNote(s:Awaited<ReturnType<typeof snapshot>>,kind:string,id:string,target:string){
 assert(s.notes.length===1&&s.notes[0]?.kind===kind&&
  s.notes[0]?.userId===target&&
  s.notes[0]?.dedupeKey===kind+":"+id+":"+target,"CANONICAL_NOTIFICATION_"+kind);
}
async function assertPending(id:string){
 const s=await snapshot(id);
 assert(s.order.status==="PENDING_APPROVAL"&&s.logs.length===0&&
  s.notes.length===0&&s.handover===null&&s.domain===0&&s.outbox===0,
  "PREMATURE_SIDE_EFFECT");
}
function noSyntheticEvents(s:Awaited<ReturnType<typeof snapshot>>){
 assert(s.domain===0&&s.outbox===0,"INVENTED_LEDGER_OR_OUTBOX");
}

  let releaseD!:()=>void,lockedD!:()=>void,queuedR!:()=>void;
  const gateD=new Promise<void>(r=>releaseD=r),readyD=new Promise<void>(r=>lockedD=r),
   readyR=new Promise<void>(r=>queuedR=r);
  const disputeFirst=db.$transaction(tx=>initiateDisputeTx(tx,{
   orderId:dFirst.id,userId:owner,reason:"real dispute first",evidencePhotos:[],
   racePoint:async()=>{lockedD();await gateD;}
  }),{maxWait:5000,timeout:15000});
  await readyD;
  let reviewerPid=0;
  const reviewLater=db.$transaction(async tx=>{
   reviewerPid=(await tx.$queryRaw<Array<{pid:number}>>`SELECT pg_backend_pid()::int AS pid`)[0]?.pid??0;
   queuedR();
   return submitRentalReviewTx(tx,{orderId:dFirst.id,userId:renter,overallRating:5});
  },{maxWait:5000,timeout:15000});
  await readyR;
  let waiting=false;
  try{
   for(let i=0;i<110;i++){
    const locks=await root.$queryRaw<Array<{blocked:boolean}>>`
      SELECT COALESCE(array_length(pg_blocking_pids(${reviewerPid}::int),1),0)>0 AS blocked
    `;
    if(locks[0]?.blocked){waiting=true;break}
    await new Promise(r=>setTimeout(r,30));
   }
   assert(waiting,"REVIEW_DISPUTE_FIRST_LOCK_NOT_OBSERVED");
  }finally{releaseD()}
  const [dw,rl]=await Promise.all([disputeFirst,reviewLater]);
  assert(succeeds(dw)&&!succeeds(rl)&&
   (await rows(dFirst.id)).length===0&&await shown(dFirst.id)===0&&
   (await root.rentalOrder.findUniqueOrThrow({where:{id:dFirst.id}})).status==="IN_DISPUTE",
   "REVIEW_DISPUTE_FIRST_STALE_REVIEW");
  console.log("[worker-cycle] PASS R3D02C07A real original dispute wins against review without creating stale review");

  // Review wins: immutable review exists before dispute, then hidden by
  // original canonical query when dispute opens from COMPLETED.
  const rFirst=await make("review-first");
  let releaseR!:()=>void,lockedR!:()=>void,queuedD!:()=>void;
  const gateR=new Promise<void>(r=>releaseR=r),readyReview=new Promise<void>(r=>lockedR=r),
   readyDispute=new Promise<void>(r=>queuedD=r);
  const reviewFirst=db.$transaction(tx=>submitRentalReviewTx(tx,{
   orderId:rFirst.id,userId:renter,overallRating:5,
   afterOrderRowLock:async()=>{lockedR();await gateR;}
  }),{maxWait:5000,timeout:15000});
  await readyReview;
  let disputePid=0;
  const disputeLater=db.$transaction(async tx=>{
   disputePid=(await tx.$queryRaw<Array<{pid:number}>>`SELECT pg_backend_pid()::int AS pid`)[0]?.pid??0;
   queuedD();
   return initiateDisputeTx(tx,{
    orderId:rFirst.id,userId:owner,reason:"real review first dispute",evidencePhotos:[]
   });
  },{maxWait:5000,timeout:15000});
  await readyDispute;
  let blocked=false;
  try{
   for(let i=0;i<110;i++){
    const locks=await root.$queryRaw<Array<{blocked:boolean}>>`
      SELECT COALESCE(array_length(pg_blocking_pids(${disputePid}::int),1),0)>0 AS blocked
    `;
    if(locks[0]?.blocked){blocked=true;break}
    await new Promise(r=>setTimeout(r,30));
   }
   assert(blocked,"REVIEW_FIRST_DISPUTE_LOCK_NOT_OBSERVED");
  }finally{releaseR()}
  const [rw,dl]=await Promise.all([reviewFirst,disputeLater]);
  assert(succeeds(rw)&&succeeds(dl)&&
   (await rows(rFirst.id)).length===1&&await shown(rFirst.id)===0&&
   (await root.rentalOrder.findUniqueOrThrow({where:{id:rFirst.id}})).status==="IN_DISPUTE",
   "REVIEW_FIRST_DISPUTE_VISIBILITY_LEAK");
  console.log("[worker-cycle] PASS R3D02C07A original review wins then dispute preserves hidden immutable review");

  const abort=await make("abort");
  let rolled=false;
  try{await db.$transaction(async tx=>{
   assert(succeeds(await submitRentalReviewTx(tx,{
    orderId:abort.id,userId:renter,overallRating:4
   })),"REVIEW_ABORT_WRITER");
   throw Error("R3D02C07A_FORCE_ABORT");
  },{maxWait:5000,timeout:15000})}catch(e){
   rolled=e instanceof Error&&e.message==="R3D02C07A_FORCE_ABORT";
  }
  assert(rolled&&(await rows(abort.id)).length===0&&
   (await notice(abort.id)).length===0&&
   (await root.rentalOrder.findUniqueOrThrow({where:{id:abort.id}})).status==="COMPLETED",
   "REVIEW_FORCED_ABORT_WRITES");
  console.log("[worker-cycle] PASS R3D02C07A original review transaction rollback after forced failure");
 }

 // R3D02C07B01: real PostgreSQL terminal snapshots + ORIGINAL canonical
 // visibleRentalReviewCondition. Controlled terminal writes are only test
 // state setup; NOT original authorized admin resolveDispute/closeDispute.
 {
  const setup=async(name:string,dual:boolean)=>{
   const id=await fixture("terminal-visibility-"+name,"IN_RENTAL");
   const completedAt=new Date();
   await root.rentalOrder.update({where:{id},data:{status:"COMPLETED",completedAt}});
   const first=await run(tx=>submitRentalReviewTx(tx,{
    orderId:id,userId:renter,overallRating:5
   }));
   assert(succeeds(first),"TERMINAL_REVIEW_FIRST");
   if(dual)assert(succeeds(await run(tx=>submitRentalReviewTx(tx,{
    orderId:id,userId:owner,overallRating:4
   }))),"TERMINAL_REVIEW_SECOND");
   const reviews=await root.rentalReview.findMany({where:{orderId:id}});
   const disputed=await run(tx=>initiateDisputeTx(tx,{
    orderId:id,userId:owner,reason:"synthetic terminal policy",evidencePhotos:[]
   }));
   assert(succeeds(disputed),"TERMINAL_DISPUTE_OPEN");
   const dispute=await root.rentalDispute.findFirstOrThrow({where:{orderId:id}});
   assert(dispute.openedFromOrderStatus==="COMPLETED"&&dispute.status==="OPEN",
    "TERMINAL_PREVIOUS_STATUS_AUTHORITY");
   assert(await root.dataHold.count({where:{
    sourceType:"RENTAL_DISPUTE",sourceId:dispute.id,status:"ACTIVE"
   }})===2,"TERMINAL_TWO_REAL_ORIGINAL_HOLDS");
   return {id,disputeId:dispute.id,reviews};
  };
  const visible=(id:string,now=new Date())=>root.rentalReview.count({
   where:visibleRentalReviewCondition({now,orderWhere:{id}})
  });
  const finalSnapshot=async(id:string,disputeId:string,action:"RESTORE_PREVIOUS"|"CLOSE_ORDER")=>
   db.$transaction(async tx=>{
    // Test-controlled coherent terminal pair, not production admin writer.
    await tx.rentalOrder.update({where:{id},data:{
     status:action==="RESTORE_PREVIOUS"?"COMPLETED":"CLOSED"
    }});
    if(action==="RESTORE_PREVIOUS"){
     const premature=await tx.rentalReview.count({where:
      visibleRentalReviewCondition({now:new Date(),orderWhere:{id}})});
     assert(premature===0,"TERMINAL_PARTIAL_RESTORE_MUST_REMAIN_HIDDEN");
    }
    await tx.rentalDispute.update({where:{id:disputeId},data:{
     status:action==="RESTORE_PREVIOUS"?"RESOLVED":"CLOSED",
     resolutionAction:action,resolvedAt:new Date()
    }});
    const changed=await tx.dataHold.updateMany({where:{
     sourceType:"RENTAL_DISPUTE",sourceId:disputeId,status:"ACTIVE"
    },data:{status:"RELEASED",releasedAt:new Date()}});
    assert(changed.count===2,"TERMINAL_SOURCE_LINKED_HOLD_SNAPSHOT");
   },{maxWait:5000,timeout:15000});
  const unchanged=async(id:string,before:Awaited<ReturnType<typeof root.rentalReview.findMany>>)=>{
   const after=await root.rentalReview.findMany({where:{orderId:id}});
   assert(before.length===after.length,"TERMINAL_REVIEW_COUNT_CHANGED");
   for(const a of before){
    const b=after.find(x=>x.id===a.id);
    assert(!!b&&b.overallRating===a.overallRating&&
     b.publishedAt?.getTime()===a.publishedAt?.getTime()&&
     b.blindUntil.getTime()===a.blindUntil.getTime(),
     "TERMINAL_IMMUTABLE_REVIEW_CHANGED");
   }
  };
  const restore=await setup("restore",true);
  const ledgerBefore=await root.domainEvent.count({where:{aggregateId:restore.id}});
  assert(restore.reviews.length===2&&
   restore.reviews.every(r=>r.publishedAt!==null)&&
   await visible(restore.id)===0,"TERMINAL_RESOLVED_PRECONDITION");
  await finalSnapshot(restore.id,restore.disputeId,"RESTORE_PREVIOUS");
  assert((await root.rentalOrder.findUniqueOrThrow({where:{id:restore.id}})).status==="COMPLETED"&&
   (await root.rentalDispute.findUniqueOrThrow({where:{id:restore.disputeId}})).status==="RESOLVED"&&
   await visible(restore.id)===2&&
   await root.dataHold.count({where:{
    sourceType:"RENTAL_DISPUTE",sourceId:restore.disputeId,status:"ACTIVE"
   }})===0&&
   await root.domainEvent.count({where:{aggregateId:restore.id}})===ledgerBefore,
   "TERMINAL_RESTORE_VISIBILITY_AND_LEDGER");
  await unchanged(restore.id,restore.reviews);
  console.log("[worker-cycle] PASS R3D02C07B01 real PG RESTORE_PREVIOUS terminal snapshot reexposes existing published reviews with no ledger drift");

  const closed=await setup("close",true);
  assert(await visible(closed.id)===0,"TERMINAL_CLOSED_PRECONDITION");
  await finalSnapshot(closed.id,closed.disputeId,"CLOSE_ORDER");
  assert((await root.rentalOrder.findUniqueOrThrow({where:{id:closed.id}})).status==="CLOSED"&&
   (await root.rentalDispute.findUniqueOrThrow({where:{id:closed.disputeId}})).status==="CLOSED"&&
   await visible(closed.id)===0,"TERMINAL_CLOSED_SHOULD_HIDE");
  await unchanged(closed.id,closed.reviews);
  console.log("[worker-cycle] PASS R3D02C07B01 real PG CLOSE_ORDER terminal snapshot keeps immutable published reviews hidden");

  const blind=await setup("blind",false);
  const future=new Date(Date.now()+8*86400000);
  assert(blind.reviews.length===1&&blind.reviews[0].publishedAt===null&&
   await visible(blind.id,future)===0,
   "TERMINAL_ACTIVE_DISPUTE_OVERRIDES_BLIND_DEADLINE");
  await finalSnapshot(blind.id,blind.disputeId,"RESTORE_PREVIOUS");
  assert(await visible(blind.id)===0&&await visible(blind.id,future)===1,
   "TERMINAL_TIME_POLICY_AFTER_RESTORE");
  await unchanged(blind.id,blind.reviews);
  console.log("[worker-cycle] PASS R3D02C07B01 original blindUntil policy remains query-time and dispute-aware after terminal restoration");
 }
}finally{await root.$disconnect();}
