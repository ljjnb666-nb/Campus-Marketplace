// #157 R3D02C01: direct original rental state machine functions in real Prisma7 PG.
import { PrismaClient } from "../generated/prisma/client.ts";
import { PrismaPg } from "@prisma/adapter-pg";
import { softDeleteExtension } from "../r3c/prisma-soft-delete.ts";
import { domainEventLedgerExtension } from "../r3c/domain-event-ledger-extension.ts";
import {
 approveRentalOrderTx,rejectRentalOrderTx,confirmPickupTx,cancelRentalOrderTx
} from "./rental-core-extracted.ts";
import {
 RENTAL_ORDER_APPROVED_KIND,RENTAL_ORDER_REJECTED_KIND,
 RENTAL_ORDER_CANCELLED_KIND,RENTAL_PICKUP_CONFIRMED_KIND
} from "./lib/notifications/notification-registry.ts";

function assert(ok:unknown,code:string):asserts ok{if(!ok)throw Error("R3D02C01_"+code);}
const url=process.env.DATABASE_URL!;
assert(new URL(url).pathname==="/"+process.env.EXPECTED_SHADOW_DB,"UNSAFE_DB");
assert(process.env.EMAIL_PROVIDER==="disabled","EXTERNAL_EMAIL_ENABLED");
const root=new PrismaClient({adapter:new PrismaPg({
 connectionString:url,max:8,connectionTimeoutMillis:2500,idleTimeoutMillis:10000
})});
const db=root.$extends(softDeleteExtension).$extends(domainEventLedgerExtension);
const owner="r3d01-seller",renter="r3d01-buyer",outsider="r3d01-outsider";
const listingId="r3d01-rental-listing";
const kinds=[RENTAL_ORDER_APPROVED_KIND,RENTAL_ORDER_REJECTED_KIND,
 RENTAL_ORDER_CANCELLED_KIND,RENTAL_PICKUP_CONFIRMED_KIND];
async function fixture(id:string,status:"PENDING_APPROVAL"|"PENDING_PICKUP"="PENDING_APPROVAL"){
 const orderId="r3d02c01-"+id;
 await root.rentalOrder.create({data:{
  id:orderId,orderNumber:"R3D02C01-"+id,rentalListingId:listingId,
  ownerId:owner,renterId:renter,startTime:new Date(Date.now()+86400000),
  endTime:new Date(Date.now()+172800000),unitPriceSnapshot:"4.00",
  pricingUnitSnapshot:"PER_DAY",rentalDuration:1,rentalAmount:"4.00",
  depositAmount:"0.00",finalAmount:"4.00",
  pickupLocationSnapshot:"isolated",returnLocationSnapshot:"isolated",status
 }});
 return orderId;
}
const run=<T>(cb:(tx:any)=>Promise<T>):Promise<T>=>db.$transaction(cb,{
 maxWait:5000,timeout:15000
}) as Promise<T>;
const approve=(id:string,userId=owner)=>run(tx=>approveRentalOrderTx(tx,{orderId:id,userId}));
const reject=(id:string,userId=owner,rejectReason="private reason never for logs")=>run(tx=>
 rejectRentalOrderTx(tx,{orderId:id,userId,rejectReason}));
const cancel=(id:string,userId=renter)=>run(tx=>cancelRentalOrderTx(tx,{
 orderId:id,userId,cancellationReason:"RENTER_CHANGED_PLAN",
 cancellationNote:"private cancellation note"
}));
const pickup=(id:string,userId:string,role:"owner"|"renter",photos:string[])=>
 run(tx=>confirmPickupTx(tx,{orderId:id,userId,role,photos}));
const succeeds=(x:{success?:boolean;error?:string})=>x.success===true;
async function snapshot(id:string){
 const [order,logs,notes,handover,domain,outbox]=await Promise.all([
  root.rentalOrder.findUniqueOrThrow({where:{id}}),
  root.rentalOrderStatusLog.findMany({where:{orderId:id}}),
  root.notification.findMany({where:{kind:{in:kinds},dedupeKey:{contains:id}}}),
  root.rentalHandoverRecord.findUnique({where:{orderId:id}}),
  root.domainEvent.count({where:{aggregateId:id}}),
  root.outboxEvent.count({where:{aggregateId:id}})
 ]);
 return {order,logs,notes,handover,domain,outbox};
}
function singleNote(s:Awaited<ReturnType<typeof snapshot>>,kind:string,id:string,target:string){
 assert(s.notes.length===1&&s.notes[0]?.kind===kind&&
  s.notes[0]?.userId===target&&
  s.notes[0]?.dedupeKey===kind+":"+id+":"+target,"CANONICAL_NOTIFICATION_"+kind);
}
async function assertPending(id:string){
 const s=await snapshot(id);
 assert(s.order.status==="PENDING_APPROVAL"&&s.logs.length===0&&
  s.notes.length===0&&s.handover===null&&s.domain===0&&s.outbox===0,
  "PREMATURE_SIDE_EFFECT");
}
function noSyntheticEvents(s:Awaited<ReturnType<typeof snapshot>>){
 assert(s.domain===0&&s.outbox===0,"INVENTED_LEDGER_OR_OUTBOX");
}
try{
 await root.$connect();
 const users=await Promise.all([owner,renter,outsider].map(id=>
  root.user.findUniqueOrThrow({where:{id}})));
 assert(users.every(u=>u.status==="ACTIVE")&&users[2]?.campusId!==users[0]?.campusId,
  "PARTICIPANT_TENANT_FIXTURE_DRIFT");
 assert((await root.rentalListing.findUniqueOrThrow({where:{id:listingId}})).ownerId===owner,
  "RENTAL_LISTING_PREREQUISITE");

 const accepted=await fixture("accepted");
 assert(succeeds(await approve(accepted)),"OWNER_APPROVE_DENIED");
 const a=await snapshot(accepted);
 assert(a.order.status==="PENDING_PICKUP"&&a.logs.length===1&&
  a.logs[0]?.fromStatus==="PENDING_APPROVAL"&&a.logs[0]?.toStatus==="PENDING_PICKUP"&&
  a.logs[0]?.operatorId===owner,"APPROVE_STATUSLOG");
 singleNote(a,RENTAL_ORDER_APPROVED_KIND,accepted,renter);
 noSyntheticEvents(a);
 assert(!succeeds(await approve(accepted)),"APPROVE_REPLAY");
 assert((await snapshot(accepted)).notes.length===1,"APPROVE_DUPLICATE_NOTIFY");

 const rejected=await fixture("rejected");
 const privateReason="NOT_FOR_STATUS_LOG_PRIVATE_REASON";
 assert(succeeds(await reject(rejected,owner,privateReason)),"REJECT_DENIED");
 const j=await snapshot(rejected);
 assert(j.order.status==="REJECTED"&&j.order.cancellationNote===privateReason&&
  j.logs.length===1&&!j.logs[0]?.note?.includes(privateReason),
  "REJECT_COPIED_FREE_TEXT");
 singleNote(j,RENTAL_ORDER_REJECTED_KIND,rejected,renter);
 noSyntheticEvents(j);
 assert(!succeeds(await reject(rejected)),"REJECT_REPLAY");
 console.log("[worker-cycle] PASS R3D02C01 original owner approve/reject canonical status log, IN_APP notification and privacy");

 const role=await fixture("roles");
 assert(!succeeds(await approve(role,renter)),"RENTER_IMPERSONATED_OWNER");
 assert(!succeeds(await reject(role,outsider)),"FOREIGN_CAMPUS_OWNER_IMPERSONATION");
 assert(!succeeds(await cancel(role,owner)),"OWNER_CANCELLED_PENDING_APPROVAL");
 assert(!succeeds(await cancel(role,outsider)),"FOREIGN_CAMPUS_CANCELLED");
 await assertPending(role);
 await root.user.update({where:{id:renter},data:{status:"SUSPENDED"}});
 let denied=false;
 try{await cancel(role,renter);}catch(e){
  denied=e instanceof Error&&(e as Error&{code?:string}).code==="AUTH_ACCOUNT_INACTIVE";
 }
 assert(denied,"SUSPENDED_RENTER_ACCEPTED");
 await assertPending(role);
 await root.user.update({where:{id:renter},data:{status:"ACTIVE"}});
 assert(succeeds(await cancel(role,renter)),"RENTER_CANCEL_DENIED");
 const can=await snapshot(role);
 assert(can.order.status==="CANCELLED"&&can.logs.length===1,
  "CANCEL_STATELOG");
 singleNote(can,RENTAL_ORDER_CANCELLED_KIND,role,owner);
 noSyntheticEvents(can);
 assert(!succeeds(await cancel(role,renter)),"CANCEL_REPLAY");
 console.log("[worker-cycle] PASS R3D02C01 original renter/owner/outsider/ACTIVE gates and cancellation replay");

 const race=await fixture("owner-decision-race");
 const [winApprove,winReject]=await Promise.all([approve(race),reject(race)]);
 assert(Number(succeeds(winApprove))+Number(succeeds(winReject))===1,
  "APPROVE_REJECT_DOUBLE_WIN");
 const decision=await snapshot(race);
 assert(decision.logs.length===1&&decision.notes.length===1&&
  (decision.order.status==="REJECTED"||decision.order.status==="PENDING_PICKUP"),
  "DECISION_FINAL_STATE");
 noSyntheticEvents(decision);
 console.log("[worker-cycle] PASS R3D02C01 original owner governance locks serialize approve vs reject");

 const pickupId=await fixture("pickup","PENDING_PICKUP");
 assert(!succeeds(await pickup(pickupId,outsider,"owner",["alien"])),
  "FOREIGN_CAMPUS_PICKUP");
 assert(!succeeds(await pickup(pickupId,renter,"owner",["spoofed"])),
  "RENTER_ROLE_SPOOF");
 assert(succeeds(await pickup(pickupId,renter,"renter",["renter-evidence"])),
  "FIRST_PICKUP");
 const partial=await snapshot(pickupId);
 assert(partial.order.status==="PENDING_PICKUP"&&
  partial.handover?.renterConfirmed===true&&
  partial.handover.ownerConfirmed===false&&
  partial.notes.length===0&&partial.logs.length===0,
  "FIRST_PICKUP_PREMATURE_COMMIT");
 assert(!succeeds(await pickup(pickupId,renter,"renter",[])),
  "FIRST_ACTOR_PHOTO_REPLACED");
 assert((await snapshot(pickupId)).handover?.photos.includes("renter-evidence"),
  "REPLAY_DELETED_EVIDENCE");
 assert(succeeds(await pickup(pickupId,owner,"owner",[])),"SECOND_PICKUP");
 const hand=await snapshot(pickupId);
 assert(hand.order.status==="IN_RENTAL"&&hand.handover?.renterConfirmed&&
  hand.handover.ownerConfirmed&&hand.logs.length===1&&
  hand.logs[0]?.toStatus==="IN_RENTAL"&&hand.notes.length===2&&
  hand.notes.every(n=>n.kind===RENTAL_PICKUP_CONFIRMED_KIND),
  "PICKUP_BOTH_AUTHORITIES");
 assert(!succeeds(await cancel(pickupId,renter)),"CANCELLED_COMPLETED_PICKUP");
 noSyntheticEvents(hand);
 console.log("[worker-cycle] PASS R3D02C01 original pickup two-party evidence, role and replay protection");

 const contested=await fixture("pickup-cancel-race","PENDING_PICKUP");
 assert(succeeds(await pickup(contested,renter,"renter",["snapshot"])),
  "RACE_SETUP_RENTER_PICKUP");
 const [pwin,cwin]=await Promise.all([
  pickup(contested,owner,"owner",[]),cancel(contested,renter)]);
 assert(Number(succeeds(pwin))+Number(succeeds(cwin))===1,
  "PICKUP_CANCEL_DOUBLE_WIN");
 const contestedResult=await snapshot(contested);
 assert(
  contestedResult.order.status==="IN_RENTAL" ?
   contestedResult.logs.length===1&&contestedResult.notes.length===2&&
   contestedResult.handover?.ownerConfirmed===true :
   contestedResult.order.status==="CANCELLED"&&
   contestedResult.logs.length===1&&contestedResult.notes.length===1&&
   contestedResult.handover?.ownerConfirmed===false,
  "PICKUP_CANCEL_INCONSISTENT_COMMIT");
 noSyntheticEvents(contestedResult);
 console.log("[worker-cycle] PASS R3D02C01 real concurrent pickup vs cancel same row lock only one winner");

 const rollback=await fixture("rollback");
 let aborted=false;
 try{await run(async tx=>{
  const success=await approveRentalOrderTx(tx,{orderId:rollback,userId:owner});
  assert(succeeds(success),"ROLLBACK_AUTHORITY_NOT_REACHED");
  assert(await tx.rentalOrderStatusLog.count({where:{orderId:rollback}})===1,
   "ROLLBACK_LOG_NOT_CREATED");
  assert(await tx.notification.count({where:{
    dedupeKey:RENTAL_ORDER_APPROVED_KIND+":"+rollback+":"+renter}})===1,
   "ROLLBACK_NOTIFICATION_NOT_CREATED");
  throw Error("R3D02C01_EXPECTED_ABORT_AFTER_NOTIFICATION");
 });}catch(e){aborted=e instanceof Error&&e.message==="R3D02C01_EXPECTED_ABORT_AFTER_NOTIFICATION";}
 assert(aborted,"FORCED_ABORT_MISSING");
 await assertPending(rollback);
 console.log("[worker-cycle] PASS R3D02C01 real rollback of original RentalOrder/StatusLog/canonical Notification");

// R3D02C02: hold the real renter cancellation after it writes CANCELLED but
// before COMMIT. The owner action must wait on the same RentalOrder row lock.
// No mocked data, no timed guess at winner, no application-layer fake locks.
for(const decision of ["approve","reject"] as const){
 const id=await fixture("cross-cancel-"+decision);
 let releaseCancel!:()=>void,signalReady!:()=>void,signalOwner!:()=>void;
 const gate=new Promise<void>(resolve=>{releaseCancel=resolve;});
 const ready=new Promise<void>(resolve=>{signalReady=resolve;});
 const ownerReady=new Promise<void>(resolve=>{signalOwner=resolve;});
 const cancelling=db.$transaction(async tx=>{
  const changed=await cancelRentalOrderTx(tx,{
   orderId:id,userId:renter,cancellationReason:"RENTER_CHANGED_PLAN"
  });
  assert(succeeds(changed),"CROSS_CANCEL_NOT_WRITTEN_"+decision);
  signalReady();
  await gate;
 },{timeout:15000,maxWait:5000});
 await ready;
 let ownerPid=0;
 const deciding=db.$transaction(async tx=>{
  const rows=await tx.$queryRaw<Array<{pid:number}>>`SELECT pg_backend_pid()::int AS pid`;
  ownerPid=rows[0]?.pid??0;
  signalOwner();
  return decision==="approve"?
   approveRentalOrderTx(tx,{orderId:id,userId:owner}):
   rejectRentalOrderTx(tx,{orderId:id,userId:owner,rejectReason:"sensitive"});
 },{timeout:15000,maxWait:5000});
 await ownerReady;
 let wasBlocked=false;
 try{
  for(let attempt=0;attempt<110;attempt++){
   const waiting=await root.$queryRaw<Array<{blocked:boolean}>>`
    SELECT COALESCE(array_length(pg_blocking_pids(${ownerPid}::int),1),0)>0 AS blocked
   `;
   if(waiting[0]?.blocked){wasBlocked=true;break;}
   await new Promise(resolve=>setTimeout(resolve,30));
  }
  assert(wasBlocked,"OWNER_ROW_LOCK_WAIT_NOT_OBSERVED_"+decision);
 }finally{releaseCancel();}
 const [ownerResult]=await Promise.all([deciding,cancelling]);
 assert(!succeeds(ownerResult),"OWNER_OVERWROTE_RENTER_CANCEL_"+decision);
 const row=await snapshot(id);
 assert(row.order.status==="CANCELLED"&&row.logs.length===1&&
  row.logs[0]?.toStatus==="CANCELLED"&&row.notes.length===1,
  "CROSS_CANCEL_STATE_LOG_NOTIFICATION_DRIFT_"+decision);
 singleNote(row,RENTAL_ORDER_CANCELLED_KIND,id,owner);
 noSyntheticEvents(row);
}
console.log("[worker-cycle] PASS R3D02C02 deterministic real PG renter CANCEL vs owner APPROVE/REJECT row-lock authority and CAS");
}finally{await root.$disconnect();}
